#!/bin/bash
set -e

echo "====== Jenkins Workspace ======"
echo $WORKSPACE

PROJECT_DIR="$WORKSPACE"            # 根目录就是项目目录
VENV_DIR="$WORKSPACE/venv"
TEST_RESULTS_DIR="$WORKSPACE/allure-results"
ALLURE_REPORT_DIR="$WORKSPACE/allure-report"

echo "====== 进入项目目录 ======"
cd "$PROJECT_DIR"

echo "====== 删除旧虚拟环境 ======"
rm -rf "$VENV_DIR"

echo "====== 创建虚拟环境 ======"
python3 -m venv "$VENV_DIR"

echo "====== 激活虚拟环境 ======"
source "$VENV_DIR/bin/activate"

echo "====== 升级 pip ======"
pip install --upgrade pip

echo "====== 安装依赖 ======"
pip install -r requirements.txt

echo "====== 设置 PYTHONPATH ======"
export PYTHONPATH=$(pwd)

echo "====== 清理旧报告 ======"
rm -rf "$TEST_RESULTS_DIR" "$ALLURE_REPORT_DIR"

echo "====== 运行 pytest ======"
pytest

echo "====== Allure 报告完成 ======"
